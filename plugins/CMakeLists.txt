# The device suite as plugins (ADR 0002, 0003). One plugin per device, all built from the
# generic adapter in src/. Codes and IDs are permanent: they must match docs/ids.md.

# The vendor-level identifiers (ADR 0024, docs/ids.md). Frozen at the first public release.
set(FD_VENDOR_NAME "Segno")
set(FD_MANUFACTURER_CODE "Sgno")
set(FD_ID_PREFIX "org.segnoaudio")
set(FD_LV2_PREFIX "https://segnoaudio.org/plugins")

# fd_add_device_plugin(ID <device-id> NAME <display> CODE <4cc> KIND effect|instrument
#                      VST3_CATEGORIES ... CLAP_FEATURES ...)
function(fd_add_device_plugin)
    cmake_parse_arguments(ARG "" "ID;NAME;CODE;KIND" "VST3_CATEGORIES;CLAP_FEATURES" ${ARGN})
    string(REPLACE "-" "_" cid "${ARG_ID}")
    set(target fd_plugin_${cid})

    set(formats VST3 LV2 Standalone)
    if(APPLE)
        list(APPEND formats AU)
    endif()
    set(is_synth FALSE)
    set(au_type kAudioUnitType_Effect)
    if(ARG_KIND STREQUAL "instrument")
        set(is_synth TRUE)
        set(au_type kAudioUnitType_MusicDevice)
    endif()

    # The name hosts show, and a file-safe one for the bundles: "&" breaks the build's own
    # commands on a bundle's path ("Phaser & Flanger" files as "Phaser and Flanger").
    set(plugin_name "${FD_VENDOR_NAME} ${ARG_NAME}")
    string(REPLACE "&" "and" product_name "${plugin_name}")
    juce_add_plugin(${target}
        PRODUCT_NAME "${product_name}"
        PLUGIN_NAME "${plugin_name}"
        COMPANY_NAME "${FD_VENDOR_NAME}"
        BUNDLE_ID "${FD_ID_PREFIX}.${ARG_ID}"
        PLUGIN_MANUFACTURER_CODE "${FD_MANUFACTURER_CODE}"
        PLUGIN_CODE "${ARG_CODE}"
        FORMATS ${formats}
        IS_SYNTH ${is_synth}
        NEEDS_MIDI_INPUT ${is_synth}
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS TRUE
        VST3_CATEGORIES ${ARG_VST3_CATEGORIES}
        AU_MAIN_TYPE ${au_type}
        LV2URI "${FD_LV2_PREFIX}/${ARG_ID}"
        COPY_PLUGIN_AFTER_BUILD FALSE)

    fd_camel_case("${ARG_ID}" camel)
    set(entry "${CMAKE_CURRENT_BINARY_DIR}/generated/${target}_entry.cpp")
    file(CONFIGURE OUTPUT "${entry}" CONTENT [[
// Generated by plugins/CMakeLists.txt. Do not edit.
#include "DeviceAudioProcessor.h"

namespace fd::devices
{
const DeviceFactory& @camel@Factory();
const ui::PanelFactory& @camel@PanelFactory();
std::span<const FactoryPresetFile> @camel@FactoryPresets();
} // namespace fd::devices

juce::AudioProcessor* JUCE_CALLTYPE createPluginFilter()
{
    return new fd::plugin::DeviceAudioProcessor(fd::devices::@camel@Factory(), &fd::devices::@camel@PanelFactory(),
                                                fd::devices::@camel@FactoryPresets());
}
]] @ONLY)

    target_sources(${target} PRIVATE "${entry}" "${CMAKE_CURRENT_SOURCE_DIR}/src/DeviceAudioProcessor.cpp")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_definitions(${target} PUBLIC
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_MODAL_LOOPS_PERMITTED=0
        $<$<PLATFORM_ID:Linux>:JUCE_JACK=1>) # the standalones (settings.md S11)
    target_link_libraries(${target}
        PRIVATE
            fd_device_${cid}_dsp fd_device_${cid}_ui fd::fd_host_juce fd::fd_device_ui
            juce::juce_audio_utils
        PUBLIC
            juce::juce_recommended_config_flags juce::juce_recommended_lto_flags fd::warnings_juce)

    clap_juce_extensions_plugin(TARGET ${target}
        CLAP_ID "${FD_ID_PREFIX}.${ARG_ID}"
        CLAP_FEATURES ${ARG_CLAP_FEATURES})
endfunction()
